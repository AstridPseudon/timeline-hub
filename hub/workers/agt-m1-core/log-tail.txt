def validate_command(command):
    try:
        json.dumps(command, allow_nan=False)
    except (ValueError, TypeError) as exc:
        raise CommandError("command must be finite JSON") from exc
    errors = sorted(_VALIDATOR.iter_errors(command), key=lambda e: str(e.path))
    if errors:
        raise CommandError(errors[0].message)
    if command['version'] != '1':
        raise CommandError("unsupported operation version", kind='unsupported_version')
    if command['op'] not in ('clip.move', 'track.create', 'timeline.inverse'):
        raise CommandError(f"unsupported operation {command['op']!r}", kind='unsupported_operation')
    operation = operations().get(command['op'])
    if operation is not None:
        errors = list(Draft202012Validator(operation.parameters).iter_errors(command['args']))
        if errors:
            raise CommandError(errors[0].message)
    if command['op'] == 'track.create':
        if command['select'] != 'parent' and not (
                isinstance(command['select'], str) and command['select'].startswith('occurrence:')
                and '/' not in command['select']):
            raise CommandError("track.create requires parent or occurrence owner")
        if command.get('intent', {}) != {}:
            raise CommandError("track.create has no intention overrides")


def _normalize(command):
    command = copy.deepcopy(command)
    validate_command(command)
    resolved = command.get('resolved', [command['select']] if command['op'] == 'clip.move' else [])
    for field in ('expect', 'authorship'):
        if field not in command:
            continue
        normalized = {}
        for key, value in command[field].items():
            ref, path = address(key, resolved)
            canonical = ref + '::' + spelling(path)
            if canonical in normalized and not equal(normalized[canonical], value):
                raise CommandError("inconsistent equivalent property guards", ref=ref, path=path)
            normalized[canonical] = value
        command[field] = normalized
    return command


def _verify_claims(ctx, command):
    for ref in command.get('created', []):
        if ctx.state.entity(ref, optional=True) is not None:
            raise CommandError('created identity is occupied', kind='identity_collision', ref=ref)
    for key, expected in command.get('expect', {}).items():
        ref, path = address(key, command.get('resolved', []))
        guard(ctx.state, ref, path, expected)
    for read in command.get('reads', []):
        guard(ctx.state, read['ref'], read['path'], read['state'])
    for selected in command.get('selections', []):
        actual = selection(ctx.state, selected['key'], selected['owner'], selected['members'])
        if not equal(actual, selected):
            raise CommandError("original selection membership/order changed", kind='selection_changed', ref=selected['owner'])

def one(rows, key, identity, *, optional=False):
    if not isinstance(rows, list):
        raise CommandError("entity collection must be an array")
    matches = [row for row in rows if isinstance(row, dict) and row.get(key) == identity]
    if len(matches) != 1:
        if optional and not matches:
            return None
        raise CommandError(f"missing or ambiguous identity {identity!r}", kind="identity_conflict")
    return matches[0]


class State:
    def __init__(self, bundle):
        self.bundle = bundle

    def occurrence(self, owner):
        return one(self.bundle["placements"], "occurrence_id", owner.removeprefix("occurrence:"))

    def config(self, owner):
        if owner == "parent":
            return self.bundle["parent"].get("config", {})
        row = self.occurrence(owner)
        return self.bundle["shots"][row["shot_id"]]["internal_timeline"]

    def collection(self, owner, kind):
        config = self.config(owner)
        if owner == "parent" and kind == "clip" and "clips" in self.bundle["parent"]:
            return self.bundle["parent"]["clips"]
        return config.get(kind + "s", [])

    def entity(self, ref, *, optional=False):
        if '/clip:' in ref or '/track:' in ref:
            owner, entity = ref.rsplit('/', 1)
            kind, identity = entity.split(':', 1)
            return one(self.collection(owner, kind), "id", identity, optional=optional)
        if ref == "parent":
            return self.bundle["parent"]
        if ref.endswith('/registry'):
            owner = ref.removesuffix('/registry')
            if owner == "parent":
                from astrid.core.timeline.parent_registry import effective_parent_registry
                return effective_parent_registry(self.bundle["parent"])
            return self.config(owner).get("registry", {})
        if ref.endswith('/source'):
            return self.config(ref.removesuffix('/source'))
        if ref.endswith('/shot'):
            return {"shot_id": self.occurrence(ref.removesuffix('/shot'))["shot_id"]}
        if ref.startswith("occurrence:") and '/' not in ref:
            return self.occurrence(ref)
        raise CommandError(f"unsupported entity reference {ref!r}")

    def get(self, ref, path):
        value = self.entity(ref, optional=True)
        if value is None:
            return copy.deepcopy(ABSENT)
        for key in path:
            if isinstance(value, dict) and key in value:
                value = value[key]
            elif isinstance(value, list) and key.isdecimal() and int(key) < len(value):
                value = value[int(key)]
            else:
                return copy.deepcopy(ABSENT)
        return present(value)

    def set(self, ref, path, state):
        if not path:
            owner, entity = ref.rsplit('/', 1)
            kind, identity = entity.split(':', 1)
            rows = self.collection(owner, kind)
            existing = self.entity(ref, optional=True)
            if existing is not None:
                index = rows.index(existing)
                if state["state"] == "absent":
                    rows.pop(index)
                else:
                    rows[index] = copy.deepcopy(state["value"])
            elif state["state"] == "present":
                rows.append(copy.deepcopy(state["value"]))
            return
        value = self.entity(ref)
        for key in path[:-1]:
            if not isinstance(value, dict) or key not in value or not isinstance(value[key], dict):
                raise CommandError("write requires recorded container creation", ref=ref, path=path)
            value = value[key]
        if not isinstance(value, dict):
            raise CommandError("property parent must be an object", ref=ref, path=path)
        if state["state"] == "absent":
            value.pop(path[-1], None)
        else:
            value[path[-1]] = copy.deepcopy(state["value"])
