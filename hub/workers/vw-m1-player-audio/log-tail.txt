+        method: 'media-element-source',
+        ok: blob.size > 0,
+        mimeType: recorder.mimeType,
+        bytes: blob.size,
+        base64: encoded,
+        tracks: destination.stream.getAudioTracks().map((track) => ({
+          kind: track.kind,
+          label: track.label,
+          enabled: track.enabled,
+          readyState: track.readyState,
+        })),
+        audioElements,
+      } satisfies CaptureResult;
+    } catch (error) {
+      return {
+        method: 'media-element-source',
+        ok: false,
+        error: error instanceof Error ? `${error.name}: ${error.message}` : String(error),
+        audioElements,
+      } satisfies CaptureResult;
+    } finally {
+      await context.close();
+    }
+  });

+async function decodeCapture(result: CaptureResult, testInfo: import('@playwright/test').TestInfo, name: string): Promise<Record<string, unknown>> {
+  if (!result.base64) return { ok: false, reason: 'capture returned no encoded recording' };
+  const recordingPath = browserEvidencePath(testInfo, `${name}.webm`);
+  const pcmPath = browserEvidencePath(testInfo, `${name}.s16le`);
+  await writeFile(recordingPath, Buffer.from(result.base64, 'base64'));
+  const { stdout: probeOutput } = await execFileAsync('ffprobe', [
+    '-v', 'error', '-show_entries', 'stream=sample_rate,channels,duration', '-of', 'json', recordingPath,
+  ]);
+  await execFileAsync('ffmpeg', [
+    '-v', 'error', '-i', recordingPath, '-ac', '2', '-ar', '44100', '-f', 's16le', pcmPath,
+  ]);
+  const pcm = await readFile(pcmPath);
+  const samples = new Int16Array(pcm.buffer, pcm.byteOffset, Math.floor(pcm.byteLength / 2));
+  const rms = Math.sqrt([...samples].reduce((sum, sample) => sum + sample * sample, 0) / Math.max(1, samples.length));
+  return {
+    ok: true,
+    containerBytes: result.bytes,
+    decodedPcmBytes: pcm.byteLength,
+    decodedStereoSamples: pcm.byteLength / 4,
+    rms,
+    ffprobe: JSON.parse(probeOutput),
+  };
+}

+            video: true,

+              enabled: track.enabled,
+              readyState: track.readyState,
+            })),

+        } catch (error) {
+          resolve({
+            method: 'getDisplayMedia',
+            ok: false,
+            error: error instanceof Error ? `${error.name}: ${error.message}` : String(error),

+  });
