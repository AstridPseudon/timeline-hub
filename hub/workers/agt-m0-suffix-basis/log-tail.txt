+def execute(state, call, *, record=False):
+    """A four-operation semantic model; these are NOT full R4 op schemas."""
+    result = copy.deepcopy(state)
+    target = call.select
+    reads, writes, bindings = set(), {}, {}
+    if call.op == "timeline.add":
+        target, = call.created
+        reads.add(key(target, "identity"))
+        if key(target, "identity") in state:
+            raise Refused("created identity collision")
+        writes = {key(target, "identity"): target,
+                  key(target, "intent.on"): {"fixed": call.args["at"]}}
+    else:
+        reads.add(key(target, "identity"))
+        if key(target, "identity") not in state:
+            raise Refused("deleted target")
+        if call.op == "clip.set-param":
+            path = key(target, call.args["path"])
+            reads.add(path)
+            writes[path] = call.args["value"]
+        elif call.op in {"clip.on", "clip.nudge"}:
+            path = key(target, "intent.on")
+            reads.add(path)
+            expression = copy.deepcopy(state[path])
+            if call.op == "clip.on":
+                moment = call.args["moment"]
+                expression = ({"anchor": ANCHOR, "offset": 0}
+                              if moment == ANCHOR else {"fixed": moment})
+            else:
+                part = "offset" if "anchor" in expression else "fixed"
+                expression[part] += call.args["seconds"]
+            if "anchor" in expression:
+                reads.add(key(expression["anchor"], "identity"))
+                bindings[ANCHOR] = expression["anchor"]
+            writes[path] = expression
+        else:
+            raise Refused("unsupported model operation")
+    expected = {path: tagged(state, path) for path in sorted(reads)}
+    if not record and (call.expect != expected or dict(call.bindings or {}) != bindings):
+        raise Refused("original guards/bindings do not match reconstructed basis")
+    result.update(copy.deepcopy(writes))
+    recorded = replace(call, expect=expected, bindings=bindings, created=call.created or (),
+                       authorship={path: "direct" for path in writes})
+    return result, recorded, reads, set(writes)

+def replay(state, calls):
+    reads, writes = set(), set()
+    for wire in calls:
+        # Real frozen dataclass/to_json; no production schema validator is called.
+        call = EditCommand(**wire)
+        state, _, consumed, emitted = execute(state, call)
+        reads.update(consumed)
+        writes.update(emitted)
+    return state, reads, writes

+    expression = state[key(target, "intent.on")]

+def request(server, base, calls, *, original=False):
+    body = {"base_publication_id": base, "base_revision_id": server.entries[base]["revision"],
+            "commands": [call.to_json() for call in calls]}
+    if original:
+        body["base_view"] = "original-result"
+    return body

+class Server:
+    """In-memory protocol specification. JSON round-trip models restart only."""
+
+    def __init__(self):
+        state = {key(CLIP, "identity"): CLIP, key(CLIP, "intent.on"): {"fixed": 0},
+                 key(CLIP, "label"): "A", key(CLIP, "opacity"): 1,
+                 key(ANCHOR, "identity"): ANCHOR, "parent::duration": 10}
+        self.entries = {"b": {"document": state, "revision": self.revision(state),
+                              "previous": None, "writes": [], "request": None}}
+        self.head, self.keys = "b", {}
+
+    @staticmethod
+    def revision(state):
+        # Content identity, intentionally independent of publication identity.
+        return json.dumps(state, sort_keys=True, separators=(",", ":"))
+
+    def basis(self, body):
+        base = body["base_publication_id"]
+        entry = self.entries[base]
+        if entry["revision"] != body["base_revision_id"]:
+            raise Refused("base pair mismatch")
+        view = body.get("base_view", "accepted")
+        if view == "accepted":
+            return copy.deepcopy(entry["document"]), base, set()
+        if view != "original-result" or entry["request"] is None:
+            raise Refused("original request/basis unavailable")
+        before, root, represented = self.basis(entry["request"])
+        after, _, _ = replay(before, entry["request"]["commands"])
+        return after, root, represented | {base}
+
+    def intervening(self, root, represented):
+        cursor, writes = self.head, set()
+        while cursor != root:
+            if cursor is None:
+                raise Refused("basis is not an ancestor")
+            entry = self.entries[cursor]
+            if cursor not in represented:
+                writes.update(entry["writes"])
+            cursor = entry["previous"]
+        return writes
+
+    def commit(self, retry_key, body):
+        encoded = json.dumps(body, sort_keys=True, separators=(",", ":"))
+        if retry_key in self.keys:
+            previous, response = self.keys[retry_key]
+            if previous != encoded:
+                raise Refused("idempotency mismatch")
+            return copy.deepcopy(response)
+        original, root, represented = self.basis(body)
+        _, reads, writes = replay(original, body["commands"])
+        if self.intervening(root, represented) & (reads | writes):
+            raise Refused("dependency written since original basis (including ABA/no-op)")
+        # Guards are checked unchanged again on the accepted head. Follow timing
+        # is derived, so a different compiled number does not replace the guard.
+        accepted, _, _ = replay(self.entries[self.head]["document"], body["commands"])
+        return self.publish(accepted, writes, body, retry_key, encoded)
+
+    def publish(self, state, writes, body=None, retry_key=None, encoded=None):
+        publication = f"p{len(self.entries)}"
+        revision = self.revision(state)
+        self.entries[publication] = {"document": copy.deepcopy(state), "revision": revision,
+                                     "previous": self.head, "writes": sorted(writes),
+                                     "request": copy.deepcopy(body)}
+        self.head = publication
+        response = {"receipt": {"receipt_id": publication},
+                    "data": {"accepted_revision_id": revision,
+                             "canonical_document": copy.deepcopy(state)}}
+        if retry_key is not None:
+            self.keys[retry_key] = [encoded, copy.deepcopy(response)]
+        return response
+
+    def agent(self, changes):
+        state = {**self.entries[self.head]["document"], **copy.deepcopy(changes)}
+        return self.publish(state, set(changes))
+
+    def read_revision(self, revision):
+        return copy.deepcopy(next(entry["document"] for entry in self.entries.values()
+                                  if entry["revision"] == revision))
+
+    def restart(self):
+        other = Server()
+        other.entries, other.head, other.keys = json.loads(json.dumps(
+            [self.entries, self.head, self.keys]))
+        return other
