+        raise CommandError('placement requires a raw clip identity', kind='identity_conflict')
+    from .commands import execute

+    ref = owner + '/clip:' + clip_id
+    return execute(bundle, [{'schema_version': 1, 'op': 'placement.move', 'version': '1',
